# expr_cases.txt - expressions for tests/analysis/test_expr.q.
# One case per line: the grammar rule, a TAB, then the AQuery text.
#   expr       an expression: operators, calls, CASE (& is AND, | is OR)
#   predicate  an expression with IN, BETWEEN, LIKE or IS [NOT] NULL
# tools/gen_type_golden.sh runs each one through the Scala type checker
# (tools/TypeDump.java) and saves its type and errors in
# tests/analysis/expr_golden.txt.
# Lines starting with # and blank lines are skipped.

# literals and names
expr	1
expr	1.5
expr	"a"
expr	TRUE
expr	FALSE
expr	2016-01-01
expr	x
expr	t.x
expr	ROWID

# & (and) and | (or): both sides boolean
expr	TRUE & FALSE
expr	x & y
expr	1 & TRUE
expr	TRUE | "a"
expr	1 | 2
expr	a > 1 & b < 2 | c = 3

# comparisons: both sides numeric
expr	1 < 2
expr	x <= y
expr	"a" < 2
expr	1 >= "b"
expr	TRUE > 1
expr	2016-01-01 < 2016-02-01

# equality: the right side has the left side's type
expr	1 = 1
expr	1 = "a"
expr	"a" = 1
expr	x = "a"
expr	"a" != y
expr	TRUE = FALSE
expr	TRUE = 1
expr	1 != 2

# arithmetic: numeric or boolean
expr	1 + 2
expr	2 * 3 / "a"
expr	x ^ TRUE
expr	x + y / z
expr	a + "x"
expr	"a" * "b"
expr	TRUE - FALSE
expr	(1 < 2) + 3

# NOT and minus
expr	NOT(TRUE)
expr	NOT(1)
expr	NOT(x)
expr	-x
expr	-FALSE
expr	-"a"
expr	-(1 + 2)

# built-in calls

# UDF calls: unknown to a checker with only built-ins

# array indexing

# CASE
expr	CASE f(c1) WHEN TRUE THEN 1 ELSE 2 END
expr	CASE f(c1) WHEN TRUE THEN 1 ELSE TRUE END
expr	CASE WHEN c1 > 2 THEN 1 ELSE 2 END
expr	CASE WHEN 1 THEN 1 END
expr	CASE max(c2) * 1 WHEN "a" THEN 1 END
expr	CASE "a" WHEN 1 THEN 2 END
expr	CASE f(c1) WHEN TRUE THEN CASE WHEN c1 > 2 THEN 1 ELSE 2 END ELSE TRUE END
expr	CASE WHEN x THEN y ELSE "a" END
expr	CASE WHEN "a" THEN 1 ELSE 2 + "b" END

# CASE with two or more WHEN branches: the Scala checker crashes
expr	CASE WHEN x THEN 1 WHEN y THEN "a" END
expr	CASE 1 WHEN 2 THEN "a" WHEN 3 THEN "b" ELSE "c" END
expr	CASE WHEN x THEN 1 WHEN y THEN 2 END
